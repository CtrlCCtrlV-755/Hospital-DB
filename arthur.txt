	TABELAS:
    
    CREATE TABLE pacientes(
        id INT AUTO_INCREMENT PRIMARY KEY,
        nome VARCHAR(100) NOT NULL,
        idade INT NOT NULL,
        cpf VARCHAR(11) UNIQUE NOT NULL,
        telefone VARCHAR(10) UNIQUE NOT NULL,
        diagnostico VARCHAR(1000) NOT NULL
        );


        CREATE TABLE pronto_socorro(
        id INT AUTO_INCREMENT PRIMARY KEY,
        nome VARCHAR(100) NOT NULL,
        telefone VARCHAR(10) NOT NULL UNIQUE,
        localização VARCHAR(100) NOT NULL
        ); 


	CREATE TABLE estoque_alimentos(
        id INT AUTO_INCREMENT PRIMARY KEY,
        quantidade INT NOT NULL,
        status BOOLEAN DEFAULT 1
        );

CREATE TABLE estacionamento (
        id INT AUTO_INCREMENT PRIMARY KEY,
        vagas VARCHAR(100) NOT NULL UNIQUE,
        entrada DATETIME NOT NULL,
        saida DATETIME NOT NULL,
        Placa VARCHAR(50) NOT NULL UNIQUE
        );

	CREATE TABLE convenios(
        id INT AUTO_INCREMENT PRIMARY KEY,
        cpf_paciente VARCHAR(11) UNIQUE NOT NULL,
        tipo_convenio VARCHAR(100),
        tempo_convenio DATETIME
        );

 CREATE TABLE produtos_limpeza_e_higiene(
        id INT AUTO_INCREMENT PRIMARY KEY,
        quantidade INT NOT NULL,
        status BOOLEAN DEFAULT 1
        );

CREATE TABLE filiais (
        id INT AUTO_INCREMENT PRIMARY KEY,
        status_hospitalares BOOLEAN DEFAULT 1,
        telefone_hospital VARCHAR(10) NOT NULL
        );

CREATE TABLE quartos (
        id INT AUTO_INCREMENT PRIMARY KEY,
        numero INT NOT NULL,
        bloco INT NOT NULL,
        status VARCHAR(100) BOOLEAN DEFAULT 1
        );
