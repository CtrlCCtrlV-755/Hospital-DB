CREATE TABLE Centro_cirurgico (
    id INT AUTO_INCREMENT PRIMARY KEY,
    numero INT NOT NULL,
    id_funcionarios INT,
    bloco INT NOT NULL,
    status VARCHAR(100),
    FOREIGN KEY (id_funcionarios) REFERENCES funcionarios(id)
);

CREATE TABLE Expediente_funcionarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cpf_funcionario VARCHAR(11) NOT NULL UNIQUE,
    entrada DATE NOT NULL,
    saida TIME NOT NULL,
    FOREIGN KEY (cpf_funcionario) REFERENCES funcionarios(cpf)
);
